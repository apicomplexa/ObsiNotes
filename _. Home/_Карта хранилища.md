# Статистика

## Ежедневная статистика

```dataviewjs
const leftSpan = (text) => `<span style='display:flex; justify-content: left; margin-top: 3px;'>${text}</span>` 
const warpWithSpan = (text) => text.split('\n').map(line => `<span>${line}</span>`).join('\n')

let statsDays = []

for (let nDaysAgo in [1,2,3,4,5,6,7]) {
	statsDays.push(DateTime.fromMillis(dv.date('today') - dv.duration(`${nDaysAgo} day`)))
} 

let naturalLangDates = statsDays.map(d => d.toLocaleString({ month: 'long', day: 'numeric'}))
naturalLangDates[0] = 'Сегодня'
naturalLangDates[1] = 'Вчера'


let todayCreatedNotes = dv.pages().where((note) => dv.equal(dv.date(note.file.cday), statsDays[0]))
let toodayModNotes = dv.pages().where((note) => dv.equal(dv.date(note.file.mday), statsDays[0]))

const statTable = dv.markdownTable(
		["" , ""], 
		[
			['Создано заметок', `<span class="text_pill red">${todayCreatedNotes.length}</span>`],
			['Отредактировано заметок', `<span class="text_pill blue">${toodayModNotes.length}</span>`]
		]
	)

let createdNotetByDays = statsDays.map(date => dv.pages().where((note) => dv.equal(dv.date(note.file.cday), date)))
let modedNotetByDays = statsDays.map(date => dv.pages().where((note) => dv.equal(dv.date(note.file.mday), date)))

const chartCode = `\`\`\`chart
type: line
labels: [${naturalLangDates.reverse()}]
series:
  - title: Coздано
    data: [${createdNotetByDays.reverse().map(i => i.length)}]
  - title: Отредактировано
	data: [${modedNotetByDays.reverse().map(i => i.length)}]
tension: 0.15
fill: false
bestFitNumber: 0
TickDisplay: false
\`\`\``
dv.paragraph(`${statTable}\n\n${chartCode}`)
```

## Последние редактированные файлы

```dataviewjs
const leftSpan = (text) => `<span style='display:flex; justify-content: left; margin-top: 3px;'>${text}</span>` 
const warpWithSpan = (text) => text.split('\n').map(line => `<span>${line}</span>`).join('\n')
const recentlyEdetedFiles = dv.pages().sort(p1 => p1.file.mtime.ts, "desc").slice(0, 15)

let reseditStr = ''

for (let group of recentlyEdetedFiles.groupBy(p => p.file.mday).sort(g => g.rows.values[0].file.mtime, "desc")) {
	let dateHeader = `${moment(group.key.ts).format("DD-MM-YYYY")}`
	dv.header(6, dateHeader)
	let recEditList = group.rows
		.sort(p1 => p1.file.mtime.ts, "desc")
		.map(p => leftSpan(`\`${moment(p.file.mtime.ts).utcOffset("+0300").format("HH:mm")}\`  ${p.file.link}`))
		.join('')
	reseditStr = reseditStr + dateHeader + '\n' + recEditList + '\n'
	dv.paragraph(recEditList + '\n')
}
```
