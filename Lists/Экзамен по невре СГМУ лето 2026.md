const header = ""
// DataviewJS: генерирует сводку задач по заголовкам
const tasks = dv.current().file.tasks.filter(t => {
    return t.header.toString().includes(header); // группировка по заголовкам
});

 " ": "Не выполнено",
 "x": "Выполнено",
 "-": "Я не буду это учить",

 ">": "Повторить",

};

const statusCounts = {};
Object.keys(statusLabels).forEach(key => statusCounts[key] = 0);

tasks.forEach(task => {
 const status = task.status in statusCounts ? task.status : "?";
 statusCounts[status]++;
});

// Общая статистика
const totalTasks = tasks.length;

// Вывод
dv.paragraph(`**Всего задач:** ${totalTasks}`);

dv.table(["Статус", "Количество"],
 Object.entries(statusCounts)
 .filter(([key, count]) => count !== 0)
 .map(([key, count]) => [ `${statusLabels[key]} (\`${key}\`)`, `${count} (${Math.trunc(count/totalTasks*100)}%)` ])
);
